    export async function retryWithBackoff<T>(run: () => Promise<T>, attempts = 3, baseMs = 200): Promise<T> {
  let last: unknown;
  for (let i = 0; i < attempts; i++) {
    try { return await run(); } catch (e) { last = e; }
    await sleep(Math.min(2000, baseMs * 2 ** i + Math.random() * baseMs));
  }
  throw last;
}
